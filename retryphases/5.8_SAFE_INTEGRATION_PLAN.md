func _generate_poisson_disk_decorations(chunk_coord: Vector2i, biome_type: int) -> Array:
    var decorations = []
    var min_distance = _get_biome_decoration_density(biome_type)
    var max_attempts = 30
    var active_list = []
    
    # Start with one seed decoration
    var seed_pos = Vector2(CHUNK_SIZE/2, CHUNK_SIZE/2)
    decorations.append({"position": seed_pos, "type": "patch"})
    active_list.append(seed_pos)
    
    while active_list.size() > 0 and decorations.size() < 50:  # Limit for performance
        var index = randi() % active_list.size()
        var current = active_list[index]
        var found = false
        
        for attempt in range(max_attempts):
            var angle = randf() * TAU
            var distance = min_distance + randf() * min_distance
            var candidate = current + Vector2.from_angle(angle) * distance
            
            # Check bounds
            if candidate.x < 0 or candidate.x >= CHUNK_SIZE or candidate.y < 0 or candidate.y >= CHUNK_SIZE:
                continue
            
            # Check distance to existing decorations
            var valid = true
            for decoration in decorations:
                if decoration.position.distance_to(candidate) < min_distance:
                    valid = false
                    break
            
            if valid:
                decorations.append({"position": candidate, "type": "patch"})
                active_list.append(candidate)
                found = true
                break
        
        if not found:
            active_list.remove_at(index)
    
    return decorations

func _get_biome_decoration_density(biome_type: int) -> float:
    # Return minimum distance between decorations based on biome
    match biome_type:

        2: return 100.0  # ICE_FIELDS - sparse
        3: return 90.0   # POISON_SWAMPS - moderate
        4: return 70.0   # CRYSTAL_CAVERNS - dense

        6: return 60.0   # DARK_FOREST - very dense

func _add_noise_based_variation(chunk_coord: Vector2i, base_color: Color) -> Color:
    # Keep original simple variation as fallback
    if not _should_use_enhanced_interpolation():
        return _add_noise_based_variation_original(chunk_coord, base_color)
    
    # Enhanced bilinear interpolation
    var world_pos = Vector2(chunk_coord.x * CHUNK_SIZE, chunk_coord.y * CHUNK_SIZE)
    
    # Sample four corners for interpolation
    var tl = _get_corner_variation(world_pos)
    var tr = _get_corner_variation(world_pos + Vector2(CHUNK_SIZE, 0))
    var bl = _get_corner_variation(world_pos + Vector2(0, CHUNK_SIZE))
    var br = _get_corner_variation(world_pos + Vector2(CHUNK_SIZE, CHUNK_SIZE))
    
    # Bilinear interpolation (center point)
    var top = tl + (tr - tl) * 0.5
    var bottom = bl + (br - bl) * 0.5
    var center_variation = top + (bottom - top) * 0.5
    
    # Apply variation to base color
    if center_variation > 0:
        return base_color.lightened(center_variation * 0.1)
    else:
        return base_color.darkened(abs(center_variation) * 0.1)

func _should_use_enhanced_interpolation() -> bool:
    # Only use if performance budget allows
    return performance_stats.average_generation_time < 60.0

func _get_corner_variation(world_pos: Vector2) -> float:
    return noise.get_noise_2d(world_pos.x * 0.001, world_pos.y * 0.001)
